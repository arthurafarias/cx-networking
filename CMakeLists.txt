cmake_minimum_required(VERSION 3.16)
project(lambdatech-networking VERSION 0.1.0)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Threads REQUIRED)

include(CMakePackageConfigHelpers)

# Header-only: lambdatech-networking produces no library binary of its own.
# Only the examples (and the self-hosted test binary below) compile to real
# object files/executables.
add_library(${PROJECT_NAME} INTERFACE)
add_library(${PROJECT_NAME}::${PROJECT_NAME} ALIAS ${PROJECT_NAME})

target_link_libraries(${PROJECT_NAME} INTERFACE Threads::Threads)

target_include_directories(${PROJECT_NAME} INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)

install(TARGETS ${PROJECT_NAME}
    EXPORT ${PROJECT_NAME}Targets
    INCLUDES DESTINATION include
)

# Install header files. testing/ subdirectories hold each module's
# colocated unit tests and are excluded so installed packages ship
# production headers only.
install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/include/
    DESTINATION include
    PATTERN "testing" EXCLUDE
)

install(EXPORT ${PROJECT_NAME}Targets
    FILE ${PROJECT_NAME}Targets.cmake
    NAMESPACE ${PROJECT_NAME}::
    DESTINATION lib/cmake/${PROJECT_NAME}
)

write_basic_package_version_file(
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}ConfigVersion.cmake
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY AnyNewerVersion
)

configure_package_config_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/CMakeTemplates/Config.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake
    INSTALL_DESTINATION lib/cmake/${PROJECT_NAME}
    PATH_VARS CMAKE_INSTALL_PREFIX
)

install(FILES
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}ConfigVersion.cmake
    DESTINATION lib/cmake/${PROJECT_NAME}
)

add_subdirectory(examples)

# Optional, external-dependency plugins - see docs.src's Plugin Folder
# Convention guideline. Each plugin/CMakeLists.txt does its own dependency
# detection and produces no targets if its library is absent.
option(LAMBDATECH_NETWORKING_BUILD_PLUGINS "Build optional external-dependency plugins under plugins/" ON)
if(LAMBDATECH_NETWORKING_BUILD_PLUGINS)
  add_subdirectory(plugins)
endif()

add_subdirectory(tools)

# Cross-stack echo benchmark (see benchmarks/). OFF by default - it pulls in
# Boost.Asio when present and is only useful for manual comparison runs.
option(LAMBDATECH_NETWORKING_BUILD_BENCHMARKS "Build the C++ side of the cross-stack benchmark under benchmarks/" OFF)
if(LAMBDATECH_NETWORKING_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks/cpp)
endif()

# Self-hosted test facility: no external test framework - the library stays
# dependency-free beyond the STL and pthreads. Each
# include/.../testing/*_test.hpp self-registers a test_group (see
# include/lambdatech/networking/testing/test_group.hpp); every one found is
# glob'd into a single generated .cpp of #includes and compiled into one
# test binary. CONFIGURE_DEPENDS re-runs the glob when a new one is added.
option(LAMBDATECH_NETWORKING_BUILD_TESTS "Build the self-hosted test facility" ON)
if(LAMBDATECH_NETWORKING_BUILD_TESTS)
  set(LAMBDATECH_NETWORKING_GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
  file(WRITE "${LAMBDATECH_NETWORKING_GENERATED_TESTS}"
    "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")

  file(GLOB_RECURSE LAMBDATECH_NETWORKING_TEST_HEADERS CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/include/*_test.hpp)
  foreach(TEST_HEADER IN LISTS LAMBDATECH_NETWORKING_TEST_HEADERS)
    file(RELATIVE_PATH RELATIVE_HEADER ${CMAKE_CURRENT_SOURCE_DIR}/include ${TEST_HEADER})
    file(APPEND "${LAMBDATECH_NETWORKING_GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
  endforeach()

  add_executable(${PROJECT_NAME}-tests
    tests/main.cpp
    "${LAMBDATECH_NETWORKING_GENERATED_TESTS}"
  )
  target_link_libraries(${PROJECT_NAME}-tests PRIVATE ${PROJECT_NAME})

  enable_testing()
  add_test(NAME ${PROJECT_NAME}-tests COMMAND ${PROJECT_NAME}-tests)

  # Coverage: instruments the test binary with --coverage (GCC/Clang only)
  # and adds a `coverage` target that produces an HTML report via gcovr,
  # published alongside the built docs site at docs/coverage/. OFF by
  # default - it forces -O0 and adds instrumentation overhead.
  option(LAMBDATECH_NETWORKING_ENABLE_COVERAGE "Instrument the test binary for gcov/gcovr coverage reporting" OFF)
  if(LAMBDATECH_NETWORKING_ENABLE_COVERAGE)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      message(FATAL_ERROR "LAMBDATECH_NETWORKING_ENABLE_COVERAGE requires GCC or Clang (--coverage), not ${CMAKE_CXX_COMPILER_ID}")
    endif()

    target_compile_options(${PROJECT_NAME}-tests PRIVATE --coverage -O0 -g)
    target_link_options(${PROJECT_NAME}-tests PRIVATE --coverage)

    find_program(LAMBDATECH_NETWORKING_GCOVR_EXECUTABLE gcovr)
    if(LAMBDATECH_NETWORKING_GCOVR_EXECUTABLE)
      set(LAMBDATECH_NETWORKING_COVERAGE_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/docs/coverage")
      add_custom_target(coverage
        COMMAND ${CMAKE_COMMAND} -E make_directory "${LAMBDATECH_NETWORKING_COVERAGE_OUTPUT_DIR}"
        COMMAND $<TARGET_FILE:${PROJECT_NAME}-tests>
        COMMAND ${LAMBDATECH_NETWORKING_GCOVR_EXECUTABLE}
                --root "${CMAKE_CURRENT_SOURCE_DIR}"
                --filter "${CMAKE_CURRENT_SOURCE_DIR}/include/lambdatech/networking/.*"
                --exclude ".*/testing/.*"
                --html --html-details
                --html-title "LambdaTech Networking Coverage"
                --sort uncovered-percent --sort-reverse
                -o "${LAMBDATECH_NETWORKING_COVERAGE_OUTPUT_DIR}/index.html"
                "${CMAKE_CURRENT_BINARY_DIR}"
        DEPENDS ${PROJECT_NAME}-tests
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        COMMENT "Running ${PROJECT_NAME}-tests under coverage instrumentation and generating an HTML report in docs/coverage"
        VERBATIM
      )
    else()
      message(WARNING "LAMBDATECH_NETWORKING_ENABLE_COVERAGE is ON but gcovr was not found - the 'coverage' target will not be available.")
    endif()
  endif()
endif()
